--========================================================--
-- DRAG FUNCTION
--========================================================--

local function MakeDraggable(Object)

	local dragging = false
	local dragStart
	local startPosition

	Object.InputBegan:Connect(function(input)

		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			dragStart = input.Position
			startPosition = Object.Position

			input.Changed:Connect(function()

				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end

			end)

		end

	end)

	UserInputService.InputChanged:Connect(function(input)

		if not dragging then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local delta = input.Position - dragStart

		Object.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)

	end)

end

local TARGET_FOLDER_NAMES = {"Enemies", "Mobs", "NPCs", "Monsters"}

local function GetRootPart()
    local Character = Player.Character
    if not Character then return nil end
    return Character:FindFirstChild("HumanoidRootPart")
end

local function FindTargetFolder()
    for _, name in ipairs(TARGET_FOLDER_NAMES) do
        local folder = workspace:FindFirstChild(name)
        if folder then return folder end
    end
    return nil
end

local function GetValidTarget(model)
    if not model or model == Player.Character then return nil end
    local humanoid = model:FindFirstChildOfClass("Humanoid")
    local root = model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChild("UpperTorso")
        or model:FindFirstChild("Torso")
    if humanoid and humanoid.Health > 0 and root then
        return humanoid, root
    end
    return nil
end

local function GetNearestTarget()
    local myRoot = GetRootPart()
    local folder = FindTargetFolder()
    if not myRoot or not folder then return nil end

    local nearest, nearestDistance
    for _, obj in ipairs(folder:GetChildren()) do
        local humanoid, root = GetValidTarget(obj)
        if humanoid and root then
            local distance = (root.Position - myRoot.Position).Magnitude
            if distance <= KillAuraRange and (not nearestDistance or distance < nearestDistance) then
                nearest = obj
                nearestDistance = distance
            end
        end
    end
    return nearest, nearestDistance
end

local function ClearTargetHighlight()
    if TargetHighlight then
        pcall(function()
            TargetHighlight:Destroy()
        end)
        TargetHighlight = nil
    end
    CurrentTarget = nil
end

local function HighlightTarget(target)
    if CurrentTarget == target and TargetHighlight then
        return
    end

    ClearTargetHighlight()

    if not target then
        return
    end

    CurrentTarget = target

    local ok, hl = pcall(function()
        local h = Instance.new("Highlight")
        h.Name = "K5_TargetHighlight"
        h.FillTransparency = 0.72
        h.OutlineTransparency = 0.05
        h.Adornee = target
        h.Parent = Main
        return h
    end)

    if ok then
        TargetHighlight = hl
    end
end

local function StartKillAura()
    if KillAuraRunning then return end
    KillAuraRunning = true

    task.spawn(function()
        while KillAura do
            local target, distance = GetNearestTarget()

            if target then
                HighlightTarget(target)
                Status.Text = ("TARGET  •  %s  •  %.0f"):format(target.Name, distance or 0)
            else
                ClearTargetHighlight()
                Status.Text = "KILL AURA  •  SEARCHING"
            end

            task.wait(0.12)
        end

        ClearTargetHighlight()
        KillAuraRunning = false
    end)
end

local function FarmCharacter()
    local Character = Player.Character
    if not Character then return nil, nil, nil end
    local Root = Character:FindFirstChild("HumanoidRootPart")

local function IsLobby()
    return workspace:FindFirstChild("MatchRoom") ~= nil
        and workspace:FindFirstChild("WorldEnemys") == nil
        and workspace:FindFirstChild("DragonEgg") == nil
end

local function GetPart(Object)
    if not Object then return nil end
    if Object:IsA("BasePart") then return Object end
    if Object:IsA("Model") then
        return Object.PrimaryPart or Object:FindFirstChild("HumanoidRootPart")
            or Object:FindFirstChildWhichIsA("BasePart", true)
    end
    return Object:FindFirstChildWhichIsA("BasePart", true)
end

local function IsAliveEnemy(Model)
    if not Model or not Model:IsA("Model") or Model == Player.Character then return false end
    local Humanoid = Model:FindFirstChildOfClass("Humanoid")
    local Root = Model:FindFirstChild("HumanoidRootPart") or Model.PrimaryPart
    return Humanoid ~= nil and Humanoid.Health > 0 and Root ~= nil
end

local function FindNearestEnemy(Root)
    local Best, BestDistance = nil, math.huge
    for _, Object in ipairs(workspace:GetDescendants()) do
        if IsAliveEnemy(Object) then
            local Part = Object:FindFirstChild("HumanoidRootPart") or Object.PrimaryPart
            local Distance = (Root.Position - Part.Position).Magnitude
            if Distance < BestDistance and Distance <= math.max(FarmRange + 60, 100) then
                Best, BestDistance = Part, Distance
            end
        end
    end
    return Best, BestDistance
end

local function FindBreakable(Root)
    -- Dragon Egg has priority in the source behavior.
    local Egg = workspace:FindFirstChild("DragonEgg")
    if Egg and not Egg:GetAttribute("Broken") then
        local Part = GetPart(Egg)
        if Part then return Part, "egg" end
    end

    local Best, BestDistance = nil, math.huge
    for _, Object in ipairs(workspace:GetChildren()) do
        if string.match(string.lower(Object.Name), "^chest") then
            local Part = GetPart(Object)
            if Part then
                local Distance = (Root.Position - Part.Position).Magnitude
                if Distance < BestDistance then
                    Best, BestDistance = Part, Distance
                end
            end
        end
    end
    if Best then return Best, "chest" end
    return nil, nil
end

local function SelectFarmTarget()
    local _, Root = FarmCharacter()
    if not Root or IsLobby() then return nil, nil end

    local Breakable, Kind = FindBreakable(Root)
    if Breakable then
        return Breakable, Kind
    end

    local Enemy = FindNearestEnemy(Root)
    if Enemy then
        return Enemy, "enemy"
    end
    return nil, nil
end

local function FindNextPortal(Root)
    local Best, BestScore, BestDistance = nil, 0, math.huge

    for _, Object in ipairs(workspace:GetDescendants()) do
        if Object:IsA("BasePart") then
            local Lower = string.lower(Object.Name)
            local ParentName = Object.Parent and string.lower(Object.Parent.Name) or ""
            local Trigger = Object:FindFirstChildOfClass("ProximityPrompt")
                or Object:FindFirstChildOfClass("TouchInterest")
                or Object:FindFirstChildOfClass("TouchTransmitter")

            if Trigger then
                local Distance = (Root.Position - Object.Position).Magnitude
                if Distance <= 600 then
                    local Score = 0
                    if string.find(Lower, "portal") or string.find(ParentName, "portal")
                        or string.find(Lower, "door") or string.find(ParentName, "door")
                        or string.find(Lower, "gate") or string.find(ParentName, "gate") then
                        Score += 10
                    elseif string.find(Lower, "next") or string.find(ParentName, "next")
                        or string.find(Lower, "exit") or string.find(ParentName, "exit")
                        or string.find(Lower, "finish") or string.find(ParentName, "finish")
                        or string.find(Lower, "teleport") or string.find(ParentName, "teleport") then
                        Score += 4
                    end

                    if Score > BestScore or (Score == BestScore and Distance < BestDistance) then
                        Best, BestScore, BestDistance = Object, Score, Distance
                    end
                end
            end
        end
    end

    return Best, BestScore
end

local function EnterNextStage()
    if not AutoProgressStage or os.clock() - LastPortalAt < 4 then return end
    local _, Root, Humanoid = FarmCharacter()
    if not Root or not Humanoid or Humanoid.Health <= 0 or IsLobby() then return end

    local Portal, Score = FindNextPortal(Root)
    if not Portal or Score < 10 then return end

    if LastPortalPosition and (Portal.Position - LastPortalPosition).Magnitude <= 3 then
        if os.clock() - LastPortalAt < 8 then return end
    end

    LastPortalAt = os.clock()
    LastPortalPosition = Portal.Position

    pcall(function()
        local Direction = Portal.Position - Root.Position
        if Direction.Magnitude < 0.1 then Direction = Root.CFrame.LookVector end
        Direction = Direction.Unit

        Root.CFrame = CFrame.new(Portal.Position - Direction * 2 + Vector3.new(0, 1, 0))
        task.wait(0.05)

        if firetouchinterest then
            firetouchinterest(Root, Portal, 0)
            task.wait(0.05)
            firetouchinterest(Root, Portal, 1)
        end

        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, game)
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftShift, false, game)
    end)

    FarmStatus = "NEXT STAGE"
end

    local Parent = Object.Parent
    while Parent and Parent ~= game do
        if Parent:IsA("GuiObject") and not Parent.Visible then return false end
        Parent = Parent.Parent
    end
    return true
end

                local Button = Object:FindFirstAncestorWhichIsA("TextButton")
                    or Object:FindFirstAncestorWhichIsA("ImageButton")
                if Button and IsVisibleGui(Button) then return Button end
            end
        elseif Object:IsA("TextButton") or Object:IsA("ImageButton") then
            local Lower = string.lower(Object.Name)
            if (string.find(Lower, "replay") or string.find(Lower, "again")
                or string.find(Lower, "restart")) and IsVisibleGui(Object) then
                return Object
            end
        end
    end
    return nil
end

local function HandleReplay()
    if not AutoReplay or os.clock() - LastReplayAt < 8 then return end
    local Button = FindReplayButton()
    if not Button then return end

    LastReplayAt = os.clock()
    FarmStatus = "REPLAY"

    pcall(function()
        if firesignal then
            firesignal(Button.MouseButton1Click)
        else
            Button:Activate()
        end
    end)
end

local function HandleDeath()
    if not AutoGiveup then return false end
    local RemainLife = Player:GetAttribute("RemainLife")
    if type(RemainLife) ~= "number" or RemainLife > 0 then return false end
    if os.clock() - LastDeathAt < 3 then return true end

    LastDeathAt = os.clock()
    FarmTarget = nil
    FarmTargetKind = nil
    FarmStatus = "DEAD / RETURN"

    local PlayerGui = Player:FindFirstChild("PlayerGui")
    local ExitButton
    if PlayerGui then
        local BattleHUD = PlayerGui:FindFirstChild("BattleHUD")
        local PlayerRevive = BattleHUD and BattleHUD:FindFirstChild("PlayerRevive")
        local ReviveFrame = PlayerRevive and PlayerRevive:FindFirstChild("ReviveFrame")
        local Revive = ReviveFrame and ReviveFrame:FindFirstChild("Revive")
        ExitButton = Revive and Revive:FindFirstChild("ExitBtn")
    end

    if ExitButton and IsVisibleGui(ExitButton) then
        pcall(function()
            if firesignal then
                firesignal(ExitButton.MouseButton1Click)
            else
                ExitButton:Activate()
            end
        end)
    end
    return true
end

-- Farm target selection / replay / stage progression.
task.spawn(function()
    while true do
        if AutoFarm then
            if HandleDeath() then
                task.wait(0.3)
            elseif IsLobby() then
                FarmTarget = nil
                FarmTargetKind = nil
                FarmStatus = "LOBBY"
                task.wait(0.5)
            else
                if os.clock() - LastFarmScan >= 0.25 then
                    LastFarmScan = os.clock()

                    if not FarmTarget or not FarmTarget.Parent then
                        FarmTarget, FarmTargetKind = SelectFarmTarget()
                    end

                    if FarmTarget then
                        FarmStatus = "FARMING • " .. tostring(FarmTargetKind)
                    else
                        HandleReplay()
                        EnterNextStage()
                        FarmStatus = "SEARCHING"
                    end
                end
                task.wait(0.08)
            end
        else
            FarmTarget = nil
            FarmTargetKind = nil
            task.wait(0.25)
        end
    end
end)

-- Orbit + melee + skill loop.
RunService.Heartbeat:Connect(function(DeltaTime)
    if not AutoFarm then return end

    local Character, Root, Humanoid = FarmCharacter()
    if not Character or not Root or not Humanoid or Humanoid.Health <= 0 or IsLobby() then return end
    if not FarmTarget or not FarmTarget.Parent then return end

    if FarmTargetKind == "enemy" then
        local TargetModel = FarmTarget.Parent
        local TargetHumanoid = TargetModel and TargetModel:FindFirstChildOfClass("Humanoid")
        if not TargetHumanoid or TargetHumanoid.Health <= 0 then
            FarmTarget = nil
            FarmTargetKind = nil
            return
        end
    end

    if LastFarmTarget ~= FarmTarget then
        LastFarmTarget = FarmTarget
        FarmPhase = 0
    end

    FarmPhase += DeltaTime * OrbitSpeed
    local TargetPosition = FarmTarget.Position
    local Offset = Vector3.new(
        math.sin(FarmPhase) * OrbitRadius,
        0,
        math.cos(FarmPhase) * OrbitRadius
    )

    local Horizontal = Vector3.new(TargetPosition.X + Offset.X, TargetPosition.Y, TargetPosition.Z + Offset.Z)
    local FinalY

    if FarmMode == "ABOVE" then
        FinalY = TargetPosition.Y + OrbitHeight
    elseif FarmMode == "GROUND" then
        FinalY = TargetPosition.Y + OrbitHeight
    else
        FinalY = TargetPosition.Y - OrbitHeight
    end

    local FinalPosition = Vector3.new(Horizontal.X, FinalY, Horizontal.Z)
    Root.CFrame = CFrame.lookAt(FinalPosition, TargetPosition)

    if FarmTargetKind == "egg" then
        if (Root.Position - FarmTarget.Position).Magnitude <= 24 then
            TriggerEgg(FarmTarget)
        end
    elseif (Root.Position - FarmTarget.Position).Magnitude <= FarmRange then
        AttackWithTool(Character)
        RunAutoSkills()
    end
end)
