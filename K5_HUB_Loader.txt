-- K5 HUB • Stable GitHub Loader
-- URL Raw tetap selama repository / branch / nama file tidak berubah.

local K5_HUB_URL = "https://raw.githubusercontent.com/frilshiaputri-cmd/K5_HUB.lua/refs/heads/main/K5_HUB_IronSoul_V6.txt"

local ok, source = pcall(function()
    return game:HttpGet(K5_HUB_URL)
end)

if not ok then
    warn("K5 HUB: gagal mengambil script dari GitHub.")
    warn(source)
    return
end

if type(source) ~= "string" or #source == 0 then
    warn("K5 HUB: file GitHub kosong atau tidak ditemukan.")
    return
end

local execute, compileError = loadstring(source)

if not execute then
    warn("K5 HUB: gagal meng-compile script.")
    warn(compileError)
    return
end

local runOk, runError = pcall(execute)

if not runOk then
    warn("K5 HUB: terjadi error saat menjalankan script.")
    warn(runError)
end
