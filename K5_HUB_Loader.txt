-- K5 HUB Loader
-- Repository: AUTO-FARM
-- Main script: K5_HUB_IronSoul_V6.txt

local URL = "https://raw.githubusercontent.com/frilshiaputri-cmd/AUTO-FARM/refs/heads/main/K5_HUB_Dungeon_Features_small_ui.txt"

local okHttp, source = pcall(function()
    return game:HttpGet(URL)
end)

if not okHttp then
    warn("[K5 HUB] Gagal mengambil script dari GitHub.")
    warn("[K5 HUB] " .. tostring(source))
    return
end

if not source or source == "" then
    warn("[K5 HUB] File utama kosong atau tidak ditemukan.")
    warn("[K5 HUB] Pastikan K5_HUB_IronSoul_V6.txt sudah ada di repository AUTO-FARM.")
    return
end

if type(loadstring) ~= "function" then
    warn("[K5 HUB] Executor tidak mendukung loadstring.")
    return
end

local compileOk, script = pcall(loadstring, source)

if not compileOk or type(script) ~= "function" then
    warn("[K5 HUB] Script utama gagal di-compile.")
    warn("[K5 HUB] " .. tostring(script))
    return
end

local runOk, err = pcall(script)

if not runOk then
    warn("[K5 HUB] Script utama gagal dijalankan.")
    warn("[K5 HUB] " .. tostring(err))
    return
end

print("[K5 HUB] Loader berhasil menjalankan K5 HUB.")
